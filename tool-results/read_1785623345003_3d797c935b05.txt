   225→function NetworkTab() {
   226→  const [nodes, setNodes] = useState<any[]>([]);
   227→  const [loading, setLoading] = useState(true);
   228→
   229→  useEffect(() => {
   230→    api("/api/federation/nodes").then((d) => setNodes(d.nodes)).catch(() => {}).finally(() => setLoading(false));
   231→  }, []);
   232→
   233→  if (loading) return <div className="flex h-full items-center justify-center"><Loader2 className="size-5 animate-spin text-primary" /></div>;
   234→
   235→  // Layout nodes in a circle with ECOWAS in center
   236→  const center = { x: 250, y: 200 };
   237→  const nationalNodes = nodes.filter((n) => n.nodeType === "national");
   238→  const regionalNodes = nodes.filter((n) => n.nodeType === "regional");
   239→
   240→  return (
   241→    <div className="h-full flex flex-col">
   242→      <div className="px-4 py-2 border-b border-border flex items-center gap-2">
   243→        <Network className="size-4 text-emerald-400" />
   244→        <span className="text-xs font-semibold">Federation Network Map</span>
   245→        <span className="text-[11px] text-muted-foreground">— {nodes.length} participating intelligence networks</span>
   246→      </div>
   247→      <div className="min-h-0 flex-1 overflow-auto gdt-scroll p-4">
   248→        <svg viewBox="0 0 500 400" className="w-full" style={{ maxHeight: "50vh" }}>
   249→          {/* Connection lines from regional center to national nodes */}
   250→          {regionalNodes.map((rn) => nationalNodes.map((nn, i) => {
   251→            const angle = (i / nationalNodes.length) * Math.PI * 2 - Math.PI / 2;
   252→            const x = center.x + Math.cos(angle) * 120;
   253→            const y = center.y + Math.sin(angle) * 120;
   254→            return <line key={`${rn.nodeId}-${nn.nodeId}`} x1={center.x} y1={center.y} x2={x} y2={y}
   255→              stroke="#71717a" strokeWidth={1} strokeOpacity={0.3} strokeDasharray="3,3" />;
   256→          }))}
   257→          {/* National nodes */}
   258→          {nationalNodes.map((n, i) => {
   259→            const angle = (i / nationalNodes.length) * Math.PI * 2 - Math.PI / 2;
   260→            const x = center.x + Math.cos(angle) * 120;
   261→            const y = center.y + Math.sin(angle) * 120;
   262→            const meta = NODE_TYPE_META[n.nodeType] ?? NODE_TYPE_META.national;
   263→            const r = 12 + n.trustScore * 10;
   264→            return (
   265→              <g key={n.nodeId}>
   266→                <circle cx={x} cy={y} r={r} fill={meta.color} fillOpacity={0.2} stroke={meta.color} strokeWidth={2} />
   267→                <text x={x} y={y + r + 12} fontSize="9" fill="#e4e4e7" textAnchor="middle" className="font-medium">{n.countryCode}</text>
   268→                <text x={x} y={y + r + 22} fontSize="7" fill="#71717a" textAnchor="middle">{n.name.split(" ")[0]}</text>
   269→                <title>{n.name} — trust {(n.trustScore * 100).toFixed(0)}%, {n.treatiesCount} treaties, {n.assetsShared} shared</title>
   270→              </g>
   271→            );
   272→          })}
   273→          {/* Regional node (ECOWAS) in center */}
   274→          {regionalNodes.map((n) => {
   275→            const meta = NODE_TYPE_META[n.nodeType] ?? NODE_TYPE_META.regional;
   276→            const r = 20 + n.trustScore * 8;
   277→            return (
   278→              <g key={n.nodeId}>
   279→                <circle cx={center.x} cy={center.y} r={r} fill={meta.color} fillOpacity={0.3} stroke={meta.color} strokeWidth={2.5} />
   280→                <text x={center.x} y={center.y + 4} fontSize="10" fill="#e4e4e7" textAnchor="middle" className="font-bold">{n.countryCode}</text>
   281→                <text x={center.x} y={center.y + r + 14} fontSize="8" fill="#a78bfa" textAnchor="middle" className="font-medium">{n.name.split(" ")[0]}</text>
   282→                <title>{n.name} — trust {(n.trustScore * 100).toFixed(0)}%</title>
   283→              </g>
   284→            );
   285→          })}
   286→        </svg>
   287→
   288→        {/* Node detail cards */}
   289→        <div className="mt-4 grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-2">
   290→          {nodes.map((n) => {
   291→            const meta = NODE_TYPE_META[n.nodeType] ?? NODE_TYPE_META.national;
   292→            return (
   293→              <div key={n.nodeId} className="rounded-lg border border-border bg-card/40 p-3">
   294→                <div className="flex items-center gap-2 mb-1">
   295→                  <span className="size-3 rounded-full" style={{ background: meta.color }} />
   296→                  <span className="text-[12px] font-semibold flex-1">{n.name}</span>
   297→                  <span className={cn("rounded px-1 py-0.5 text-[9px] font-medium",
   298→                    n.status === "active" ? "bg-emerald-500/10 text-emerald-300" : "bg-rose-500/10 text-rose-300")}>{n.status}</span>
   299→                </div>
   300→                <div className="grid grid-cols-2 gap-1 text-[9px] mt-1.5">
   301→                  <div className="rounded bg-card/30 px-1.5 py-1"><span className="text-muted-foreground">Trust</span> <span className="font-mono font-semibold" style={{ color: meta.color }}>{(n.trustScore * 100).toFixed(0)}%</span></div>
   302→                  <div className="rounded bg-card/30 px-1.5 py-1"><span className="text-muted-foreground">Treaties</span> <span className="font-mono font-semibold">{n.treatiesCount}</span></div>
   303→                  <div className="rounded bg-card/30 px-1.5 py-1"><span className="text-muted-foreground">Shared</span> <span className="font-mono font-semibold">{n.assetsShared}</span></div>
   304→                  <div className="rounded bg-card/30 px-1.5 py-1"><span className="text-muted-foreground">Consumed</span> <span className="font-mono font-semibold">{n.assetsConsumed}</span></div>
   305→                </div>
   306→                {n.lastSeenAt && <div className="text-[9px] text-muted-foreground mt-1">Last seen {timeAgo(n.lastSeenAt)}</div>}
   307→              </div>
   308→            );
   309→          })}
   310→        </div>
   311→      </div>
   312→    </div>
   313→  );
   314→}
