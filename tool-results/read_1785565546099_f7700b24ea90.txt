    21→  offset?: number;
    22→} = {}): Promise<EntitiesResponse> {
    23→  const sp = new URLSearchParams();
    24→  if (params.kind) sp.set("kind", params.kind);
    25→  if (params.regionId) sp.set("regionId", params.regionId);
    26→  if (params.bbox) sp.set("bbox", params.bbox);
    27→  if (params.limit) sp.set("limit", String(params.limit));
    28→  if (params.offset) sp.set("offset", String(params.offset));
    29→  const res = await fetch(`/api/world-model/entities?${sp}`);
    30→  if (!res.ok) throw new Error(`fetchEntities: ${res.status}`);
    31→  return res.json();
    32→}

   850→// ---- React hooks ----
   851→
   852→export function useAsync<T>(fn: () => Promise<T>, deps: any[] = []): { data: T | null; loading: boolean; error: string | null; refresh: () => void } {
   853→  const [state, setState] = useState<{ data: T | null; loading: boolean; error: string | null }>({
   854→    data: null,
   855→    loading: true,
   856→    error: null,
   857→  });
   858→  const [nonce, setNonce] = useState(0);
   859→
   860→  const refresh = useCallback(() => setNonce((n) => n + 1), []);
   861→
   862→  useEffect(() => {
   863→    let active = true;
   864→    // mark loading for this fetch round (async — avoids synchronous setState in effect body)
   865→    Promise.resolve().then(() => {
   866→      if (active) setState((s) => ({ ...s, loading: true, error: null }));
   867→    });
   868→    fn()
   869→      .then((d) => {
   870→        if (active) setState({ data: d, loading: false, error: null });
   871→      })
   872→      .catch((e) => {
   873→        if (active)
   874→          setState({ data: null, loading: false, error: e instanceof Error ? e.message : String(e) });
   875→      });
   876→    return () => {
   877→      active = false;
   878→    };
   879→  }, [...deps, nonce]);
   880→
   881→  return { ...state, refresh };
   882→}
