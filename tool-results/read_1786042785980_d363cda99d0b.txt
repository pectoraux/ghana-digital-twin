    32→export async function depositCredits(accountId: string, amount: number, description: string, referenceType?: string, referenceId?: string): Promise<any> {
    33→  const account = await db.creditAccount.findUnique({ where: { accountId } });
    34→  if (!account) throw new Error(`Account ${accountId} not found`);
    35→
    36→  const txId = `CTX-${Date.now().toString(36).toUpperCase()}-${Math.random().toString(36).slice(2, 5).toUpperCase()}`;
    37→  const tx = await db.creditTransaction.create({
    38→    data: {
    39→      transactionId: txId,
    40→      toAccountId: accountId,
    41→      toOwnerName: account.ownerName,
    42→      amount,
    43→      type: "deposit",
    44→      referenceType: referenceType ?? null,
    45→      referenceId: referenceId ?? null,
    46→      description,
    47→    },
    48→  });
    49→
    50→  await db.creditAccount.update({
    51→    where: { id: account.id },
    52→    data: {
    53→      balance: { increment: amount },
    54→      totalDeposited: { increment: amount },
    55→    },
    56→  });
    57→
    58→  return serializeCreditTx(tx);
    59→}

    61→export async function transferCredits(fromAccountId: string, toAccountId: string, amount: number, type: string, description: string, referenceType?: string, referenceId?: string): Promise<any> {
    62→  const from = await db.creditAccount.findUnique({ where: { accountId: fromAccountId } });
    63→  const to = await db.creditAccount.findUnique({ where: { accountId: toAccountId } });
    64→  if (!from || !to) throw new Error("Account not found");
    65→  if (from.balance < amount) throw new Error(`Insufficient balance: ${from.balance} < ${amount}`);
    66→
    67→  const txId = `CTX-${Date.now().toString(36).toUpperCase()}-${Math.random().toString(36).slice(2, 5).toUpperCase()}`;
    68→  const tx = await db.creditTransaction.create({
    69→    data: {
    70→      transactionId: txId,
    71→      fromAccountId,
    72→      toAccountId,
    73→      fromOwnerName: from.ownerName,
    74→      toOwnerName: to.ownerName,
    75→      amount,
    76→      type, // purchase | reward | royalty | platform_fee
    77→      referenceType: referenceType ?? null,
    78→      referenceId: referenceId ?? null,
    79→      description,
    80→    },
    81→  });
    82→
    83→  await db.creditAccount.update({ where: { id: from.id }, data: { balance: { decrement: amount }, totalSpent: { increment: amount } } });
    84→  await db.creditAccount.update({ where: { id: to.id }, data: { balance: { increment: amount }, totalEarned: { increment: amount } } });
    85→
    86→  return serializeCreditTx(tx);
    87→}

   203→export async function createLineage(input: {
   204→  assetId: string;
   205→  parentAssetIds: string[];
   206→  derivationType: string; // derived_from | fused_with | enhanced_by | verified_by
   207→  lineageChain?: { assetId: string; producerId: string; role: string; depth: number }[];
   208→}): Promise<any> {
   209→  // Compute depth = max parent depth + 1
   210→  let maxParentDepth = -1;
   211→  const chain: any[] = input.lineageChain ?? [];
   212→  for (const parentId of input.parentAssetIds) {
   213→    const parentLineage = await db.assetLineage.findUnique({ where: { assetId: parentId } }).catch(() => null);
   214→    if (parentLineage && parentLineage.depth > maxParentDepth) maxParentDepth = parentLineage.depth;
   215→    // Add parent to chain if not already there
   216→    const parentAsset = await db.intelligenceAsset.findUnique({ where: { assetId: parentId } }).catch(() => null);
   217→    if (parentAsset) {
   218→      chain.push({ assetId: parentId, producerId: parentAsset.producerId, role: "parent", depth: (parentLineage?.depth ?? 0) });
   219→    }
   220→  }
   221→  const depth = maxParentDepth + 1;
   222→  // Add self to chain
   223→  const selfAsset = await db.intelligenceAsset.findUnique({ where: { assetId: input.assetId } }).catch(() => null);
   224→  if (selfAsset) chain.push({ assetId: input.assetId, producerId: selfAsset.producerId, role: "self", depth });
   225→
   226→  const lineageId = `LIN-${Date.now().toString(36).toUpperCase()}-${Math.random().toString(36).slice(2, 5).toUpperCase()}`;
   227→  const lineage = await db.assetLineage.create({
   228→    data: {
   229→      lineageId,
   230→      assetId: input.assetId,
   231→      parentAssetIds: JSON.stringify(input.parentAssetIds),
   232→      lineageChain: JSON.stringify(chain),
   233→      derivationType: input.derivationType,
   234→      depth,
   235→    },
   236→  });
   237→
   238→  return serializeLineage(lineage);
   239→}

   241→/**
   242→ * Distribute royalties when a derivative asset generates value.
   243→ * Walks the lineage chain and distributes royalties to ancestor producers
   244→ * based on their royalty rate × depth decay.
   245→ */
   246→export async function distributeRoyalties(sourceAssetId: string, sourceTransactionId: string, totalValue: number): Promise<any> {
   247→  const lineage = await db.assetLineage.findUnique({ where: { assetId: sourceAssetId } }).catch(() => null);
   248→  if (!lineage) return null;
   249→
   250→  const chain = JSON.parse(lineage.lineageChain || "[]") as any[];
   251→  const ancestors = chain.filter((c) => c.role === "parent" || c.depth < lineage.depth);
   252→
   253→  // Get licenses for each ancestor to find royalty rates
   254→  const recipients: { assetId: string; producerId: string; amount: number; royaltyRate: number; depth: number }[] = [];
   255→  let remainingValue = totalValue;
   256→
   257→  for (const ancestor of ancestors.sort((a, b) => a.depth - b.depth)) {
   258→    if (remainingValue <= 0) break;
   259→    const license = await db.intelligenceLicense.findFirst({ where: { assetId: ancestor.assetId, status: "active" } }).catch(() => null);
   260→    const royaltyRate = license?.royaltyRate ?? 0.05;
   261→    // Depth decay: deeper ancestors get less (depth^0.5 decay)
   262→    const decayFactor = 1 / Math.sqrt(ancestor.depth + 1);
   263→    const amount = totalValue * royaltyRate * decayFactor;
   264→    if (amount > 0) {
   265→      recipients.push({ assetId: ancestor.assetId, producerId: ancestor.producerId, amount, royaltyRate, depth: ancestor.depth });
   266→      remainingValue -= amount;
   267→    }
   268→  }
   269→
   270→  const royaltyId = `ROY-${Date.now().toString(36).toUpperCase()}-${Math.random().toString(36).slice(2, 5).toUpperCase()}`;
   271→  const totalRoyaltyAmount = recipients.reduce((s, r) => s + r.amount, 0);
   272→
   273→  const royalty = await db.royaltyDistribution.create({
   274→    data: {
   275→      royaltyId,
   276→      sourceTransactionId,
   277→      sourceAssetId,
   278→      totalRoyaltyAmount,
   279→      royaltyRecipients: JSON.stringify(recipients),
   280→    },
   281→  });
   282→
   283→  // Distribute credits to ancestor producers
   284→  for (const r of recipients) {
   285→    const account = await getOrCreateAccount(r.producerId, "citizen", r.producerId).catch(() => null);
   286→    if (account) {
   287→      // Mint royalty credits (platform funds)
   288→      await depositCredits(account.accountId, r.amount, `Royalty from derivative asset ${sourceAssetId} (depth ${r.depth})`, "lineage_royalty", royaltyId).catch(() => null);
   289→    }
   290→  }
   291→
   292→  return serializeRoyalty(royalty);
   293→}

   309→export async function computeProducerMarketScore(producerId: string): Promise<any> {
   310→  // Gather data
   311→  const assets = await db.intelligenceAsset.findMany({ where: { producerId } });
   312→  const citizen = await db.citizen.findUnique({ where: { citizenId: producerId } }).catch(() => null);
   313→  const trustScore = await db.trustScore.findUnique({ where: { citizenId: producerId } }).catch(() => null);
   314→  const creditAccount = await db.creditAccount.findFirst({ where: { ownerId: producerId } }).catch(() => null);
   315→
   316→  const trust = trustScore?.propagatedTrust ?? citizen?.civicScore ?? 50;
   317→  const civic = citizen?.civicScore ?? 50;
   318→  const avgIqs = assets.length > 0 ? assets.reduce((s, a) => s + a.iqs, 0) / assets.length : 0.5;
   319→  const iqsContribution = avgIqs * 100;
   320→  const valueCreated = creditAccount?.totalEarned ?? 0;
   321→
   322→  // Composite: trust 25% + civic 20% + IQS 25% + valueCreated 30% (normalized)
   323→  const valueScore = Math.min(100, (valueCreated / 1000) * 100); // 1000 credits = max
   324→  const marketScore = trust * 0.25 + civic * 0.20 + iqsContribution * 0.25 + valueScore * 0.30;
   325→
   326→  let tier = "amateur";
   327→  if (marketScore >= 85) tier = "elite";
   328→  else if (marketScore >= 70) tier = "expert";
   329→  else if (marketScore >= 55) tier = "professional";
   330→  else if (marketScore >= 40) tier = "emerging";
   331→
   332→  const totalSales = assets.reduce((s, a) => s + a.purchaseCount, 0);
   333→  const avgRating = assets.reduce((s, a) => s + a.ratingAvg, 0) / Math.max(1, assets.filter((a) => a.ratingCount > 0).length);
   334→
   335→  const row = await db.producerMarketScore.upsert({
   336→    where: { producerId },
   337→    update: {
   338→      producerType: citizen ? "citizen" : "agent",
   339→      producerName: citizen?.handle ?? producerId,
   340→      trustScore: trust, civicScore: civic, iqsContribution, valueCreated,
   341→      marketScore, marketTier: tier,
   342→      totalAssets: assets.length, totalSales, avgRating: avgRating || 0,
   343→      lastUpdated: new Date(),
   344→    },
   345→    create: {
   346→      producerId,
   347→      producerType: citizen ? "citizen" : "agent",
   348→      producerName: citizen?.handle ?? producerId,
   349→      trustScore: trust, civicScore: civic, iqsContribution, valueCreated,
   350→      marketScore, marketTier: tier,
   351→      totalAssets: assets.length, totalSales, avgRating: avgRating || 0,
   352→    },
   353→  });
   354→
   355→  // Compute rank
   356→  const allScores = await db.producerMarketScore.findMany({ orderBy: { marketScore: "desc" } });
   357→  const rank = allScores.findIndex((s) => s.producerId === producerId) + 1;
   358→  await db.producerMarketScore.update({ where: { id: row.id }, data: { marketRank: rank } });
   359→
   360→  return { ...serializeProducerScore(row), marketRank: rank };
   361→}

   392→export async function recordAgentRevenue(agentId: string, agentName: string, amount: number, description: string): Promise<any> {
   393→  const revenue = await getOrCreateAgentRevenue(agentId, agentName);
   394→  const developerAmt = amount * revenue.developerShare;
   395→  const trainingAmt = amount * revenue.trainingDataShare;
   396→  const infraAmt = amount * revenue.infrastructureShare;
   397→  const platformAmt = amount * revenue.platformShare;
   398→
   399→  await db.agentRevenue.update({
   400→    where: { agentId },
   401→    data: {
   402→      totalRevenue: { increment: amount },
   403→      monthlyRevenue: { increment: amount },
   404→      developerPaid: { increment: developerAmt },
   405→      trainingDataPaid: { increment: trainingAmt },
   406→      infrastructurePaid: { increment: infraAmt },
   407→      platformPaid: { increment: platformAmt },
   408→      totalOutputs: { increment: 1 },
   409→      lastEarnedAt: new Date(),
   410→    },
   411→  });
   412→
   413→  // Transfer credits to agent's account, then allocate
   414→  const agentAccount = await getOrCreateAccount(agentId, "agent", agentName);
   415→  await depositCredits(agentAccount.accountId, amount, description, "agent_revenue", agentId).catch(() => null);
   416→
   417→  return { amount, allocation: { developer: developerAmt, trainingData: trainingAmt, infrastructure: infraAmt, platform: platformAmt } };
   418→}

   487→export async function submitPrediction(input: {
   488→  policyId: string;
   489→  predictorId: string;
   490→  predictorType: string;
   491→  predictorName: string;
   492→  predictedOutcome: string; // confirmed | denied | partial
   493→  confidence?: number;
   494→  reasoning?: string;
   495→  trustScore?: number;
   496→}): Promise<any> {
   497→  const policy = await db.insurancePolicy.findUnique({ where: { policyId: input.policyId } });
   498→  if (!policy) throw new Error(`Policy ${input.policyId} not found`);
   499→  if (policy.status !== "open") throw new Error(`Policy is ${policy.status}`);
   500→
   501→  const predictionId = await nextPredictionId();
   502→  const pred = await db.insurancePrediction.create({
   503→    data: {
   504→      predictionId,
   505→      policyId: input.policyId,
   506→      predictorId: input.predictorId,
   507→      predictorType: input.predictorType,
   508→      predictorName: input.predictorName,
   509→      predictedOutcome: input.predictedOutcome,
   510→      confidence: input.confidence ?? 0.5,
   511→      reasoning: input.reasoning ?? null,
   512→      trustScore: input.trustScore ?? 0.5,
   513→    },
   514→  });
   515→
   516→  await db.insurancePolicy.update({ where: { policyId: input.policyId }, data: { predictionCount: { increment: 1 } } });
   517→  return serializePrediction(pred);
   518→}

   527→/**
   528→ * Resolve an insurance policy — computes accuracy scores and distributes payouts.
   529→ * Accuracy = how close the prediction was to the actual outcome.
   530→ */
   531→export async function resolvePolicy(policyId: string, outcome: string, notes: string): Promise<any> {
   532→  const policy = await db.insurancePolicy.findUnique({ where: { policyId } });
   533→  if (!policy) throw new Error(`Policy ${policyId} not found`);
   534→
   535→  const predictions = await db.insurancePrediction.findMany({ where: { policyId } });
   536→  let totalAccuracy = 0;
   537→  const updates: any[] = [];
   538→
   539→  for (const pred of predictions) {
   540→    // Accuracy: 1.0 if exact match, 0.5 if partial match, 0.0 if wrong
   541→    let accuracy = 0;
   542→    if (pred.predictedOutcome === outcome) accuracy = 1.0;
   543→    else if (outcome === "partial" || pred.predictedOutcome === "partial") accuracy = 0.5;
   544→
   545→    // Weight by confidence + trust
   546→    const weightedAccuracy = accuracy * (0.5 + pred.confidence * 0.3 + pred.trustScore * 0.2);
   547→    totalAccuracy += weightedAccuracy;
   548→
   549→    updates.push({ pred, accuracy, weightedAccuracy });
   550→  }
   551→
   552→  // Distribute payout proportionally to weighted accuracy
   553→  const payouts: { predictionId: string; predictorId: string; amount: number; accuracy: number }[] = [];
   554→  for (const u of updates) {
   555→    const share = totalAccuracy > 0 ? u.weightedAccuracy / totalAccuracy : 0;
   556→    const payout = policy.totalPayout * share;
   557→    await db.insurancePrediction.update({ where: { id: u.pred.id }, data: { accuracyScore: u.accuracy, payoutAmount: payout } });
   558→
   559→    if (payout > 0) {
   560→      const account = await getOrCreateAccount(u.pred.predictorId, u.pred.predictorType, u.pred.predictorName).catch(() => null);
   561→      if (account) {
   562→        await depositCredits(account.accountId, payout, `Insurance payout for ${policyId} (accuracy ${(u.accuracy * 100).toFixed(0)}%)`, "insurance", u.pred.predictionId).catch(() => null);
   563→      }
   564→      payouts.push({ predictionId: u.pred.predictionId, predictorId: u.pred.predictorId, amount: payout, accuracy: u.accuracy });
   565→    }
   566→  }
   567→
   568→  await db.insurancePolicy.update({
   569→    where: { policyId },
   570→    data: { status: "resolved", outcome, outcomeNotes: notes, resolvedAt: new Date() },
   571→  });
   572→
   573→  return { policyId, outcome, predictionsResolved: predictions.length, totalPayout: policy.totalPayout, payouts };
   574→}
