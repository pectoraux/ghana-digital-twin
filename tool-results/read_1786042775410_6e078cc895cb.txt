   356→model Observation {
   357→  id              String   @id @default(uuid())
   358→  uuid            String   @unique
   359→  type            String   // surface_disturbance | water_body_change | vegetation_loss | burn_event | moisture_stress
   360→  title           String
   361→  summary         String   // objective description, no legal conclusions
   362→  // spatial
   363→  geometryGeoJson String   // polygon or point
   364→  centroidLng     Float
   365→  centroidLat     Float
   366→  minLng          Float
   367→  minLat          Float
   368→  maxLng          Float
   369→  maxLat          Float
   370→  areaHa          Float    // affected area in hectares
   371→  // temporal
   372→  observedAt      DateTime
   373→  firstObserved   DateTime
   374→  lastUpdated     DateTime
   375→  durationDays    Float?   // how long the change has persisted
   376→  // assessment (fused)
   377→  confidence      Float    // 0..1, fused from all evidence
   378→  uncertainty     Float    // ± propagated uncertainty
   379→  status          String   @default("active") // candidate | active | monitoring | resolved
   380→  severity        String   @default("moderate") // low | moderate | high | critical

   388→  createdAt       DateTime @default(now())

   400→  @@index([type])
   401→  @@index([status])
   402→  @@index([severity])
